    /**
     * Implicit QR algorithm with Francis double-shift for Hessenberg matrices.
     * Converges the Hessenberg matrix H to real Schur form T = Q^T H Q,
     * then extracts eigenvalues and eigenvectors from T.
     */
    private Tuple2<IDoubleVector, IDoubleMatrix> implicitQRAlgorithm(double[][] H) {
        int n = H.length;
        double[][] Q = new double[n][n];
        double[][] T = new double[n][n];

        for (int i = 0; i < n; i++) {
            System.arraycopy(H[i], 0, T[i], 0, n);
            Q[i][i] = 1.0;
        }

        final int maxIter = maxIterations;
        final double eps = epsilon;

        int activeN = n;
        for (int iter = 0; iter < maxIter; iter++) {
            // Deflation from bottom: look for negligible subdiagonal entries.
            while (activeN > 2) {
                int i = activeN - 1;
                double s = Math.abs(T[i - 1][i - 1]) + Math.abs(T[i][i]);
                if (s < eps) s = 1.0;
                if (Math.abs(T[i][i - 1]) <= eps * s) {
                    T[i][i - 1] = 0.0;
                    activeN--;
                    continue;
                }
                // Check for converged 2x2 block one row higher
                if (activeN > 2) {
                    int j = activeN - 2;
                    double s2 = Math.abs(T[j - 1][j - 1]) + Math.abs(T[j][j]);
                    if (s2 < eps) s2 = 1.0;
                    if (Math.abs(T[j][j - 1]) <= eps * s2) {
                        T[j][j - 1] = 0.0;
                        activeN = j;
                        continue;
                    }
                }
                break;
            }

            if (activeN <= 1) break;

            if (activeN == 2) {
                double a = T[n - 2][n - 2], b = T[n - 2][n - 1];
                double c = T[n - 1][n - 2], d = T[n - 1][n - 1];
                double tr = a + d, dt = a * d - b * c;
                double disc = tr * tr - 4.0 * dt;
                if (disc >= 0) {
                    double sd = Math.sqrt(disc);
                    T[n - 2][n - 2] = (tr + sd) / 2.0;
                    T[n - 1][n - 1] = (tr - sd) / 2.0;
                }
                T[n - 2][n - 1] = 0.0;
                T[n - 1][n - 2] = 0.0;
                break;
            }

            performFrancisQRStep(T, Q, activeN);
            cleanRoundingErrors(T, n);
        }

        double[] eigenvalues = extractRealSchurEigenvalues(T, n, eps);
        double[][] eigvecsRaw = backSubstituteSchurEigenvectors(T, Q, n, eps);

        normalizeEigenvectors(eigvecsRaw, n);
        sortEigenvaluesAndVectors(eigenvalues, eigvecsRaw);

        return new Tuple2<>(IDoubleVector.of(eigenvalues), (IDoubleMatrix) Linalg.matrix(eigvecsRaw));
    }

    // ======================== Francis double-shift QR step ========================

    /**
     * One Francis double-shift QR step on the active submatrix T[0:activeN][0:activeN].
     *
     * Reference: Golub & Van Loan, Matrix Computations, 4th ed., Algorithm 7.5.1.
     */
    private void performFrancisQRStep(double[][] T, double[][] Q, int activeN) {
        int n = Q.length;

        // Shifts = eigenvalues of trailing 2x2 of the active submatrix
        double a = T[activeN - 2][activeN - 2];
        double b = T[activeN - 2][activeN - 1];
        double c = T[activeN - 1][activeN - 2];
        double d = T[activeN - 1][activeN - 1];
        double traceS = a + d;
        double detS   = a * d - b * c;

        // First 3 entries of (T^2 - traceS*T + detS*I) * e1
        double h00 = T[0][0], h01 = T[0][1];
        double h10 = T[1][0], h11 = T[1][1];
        double h21 = (activeN > 2) ? T[2][1] : 0.0;

        double x = h00 * h00 + h01 * h10 - traceS * h00 + detS;
        double y = h10 * (h00 + h11 - traceS);
        double z = h10 * h21;

        for (int k = 0; k < activeN - 1; k++) {
            double[] hh;
            boolean use3 = (k < activeN - 2) && (Math.abs(z) > 1e-30);
            if (use3) {
                hh = householder3(x, y, z);
            } else {
                hh = householder2(x, y);
            }

            int hvLen = hh.length;
            double beta = hh[hvLen - 1];
            if (beta == 0.0) {
                if (k < activeN - 2) {
                    x = T[k + 1][k];
                    y = T[k + 2][k];
                    z = (k + 3 < activeN) ? T[k + 3][k] : 0.0;
                }
                continue;
            }

            int rEnd = Math.min(k + hvLen, activeN) - 1;
            int cEnd = k + hvLen - 2;

            // Left multiply: T(k:rEnd, max(0,k-1):activeN-1) = P * T(...)
            int leftColStart = (k > 0) ? k - 1 : 0;
            applyHouseholderLeft(T, hh, beta, k, rEnd, leftColStart, activeN - 1);

            // Right multiply: T(0:maxRow, k:cEnd) = T(...) * P^T
            int maxRow = Math.min(k + hvLen + 1, activeN) - 1;
            applyHouseholderRight(T, hh, beta, 0, maxRow, k, cEnd);

            // Accumulate: Q(0:n-1, k:cEnd) = Q(...) * P^T
            applyHouseholderRight(Q, hh, beta, 0, n - 1, k, cEnd);

            if (k < activeN - 2) {
                x = T[k + 1][k];
                y = T[k + 2][k];
                z = (k + 3 < activeN) ? T[k + 3][k] : 0.0;
            }
        }
    }

    // ======================== Householder helpers ========================

    private static double[] householder3(double x, double y, double z) {
        double norm = Math.sqrt(x * x + y * y + z * z);
        if (norm == 0.0) return new double[]{0.0, 0.0, 0.0};
        double alpha = (x > 0) ? -norm : norm;
        double beta = (alpha - x) / alpha;
        double v1 = y / (x - alpha);
        double v2 = z / (x - alpha);
        return new double[]{v1, v2, beta};
    }

    private static double[] householder2(double x, double y) {
        double norm = Math.sqrt(x * x + y * y);
        if (norm == 0.0) return new double[]{0.0, 0.0};
        double alpha = (x > 0) ? -norm : norm;
        double beta = (alpha - x) / alpha;
        double v1 = y / (x - alpha);
        return new double[]{v1, beta};
    }

    private static void applyHouseholderLeft(double[][] A, double[] hh, double beta,
                                             int rStart, int rEnd, int cStart, int cEnd) {
        int hvLen = rEnd - rStart + 1;
        double[] v = new double[hvLen];
        v[0] = 1.0;
        for (int t = 1; t < hvLen; t++) v[t] = hh[t - 1];
        for (int j = cStart; j <= cEnd; j++) {
            double s = 0.0;
            for (int t = 0; t < hvLen; t++) s += v[t] * A[rStart + t][j];
            s *= beta;
            for (int t = 0; t < hvLen; t++) A[rStart + t][j] -= s * v[t];
        }
    }

    private static void applyHouseholderRight(double[][] A, double[] hh, double beta,
                                              int rStart, int rEnd, int cStart, int cEnd) {
        int hvLen = cEnd - cStart + 1;
        double[] v = new double[hvLen];
        v[0] = 1.0;
        for (int t = 1; t < hvLen; t++) v[t] = hh[t - 1];
        for (int i = rStart; i <= rEnd; i++) {
            double s = 0.0;
            for (int t = 0; t < hvLen; t++) s += v[t] * A[i][cStart + t];
            s *= beta;
            for (int t = 0; t < hvLen; t++) A[i][cStart + t] -= s * v[t];
        }
    }

    // ======================== Eigenvalue / Eigenvector extraction ========================

    private double[] extractRealSchurEigenvalues(double[][] T, int n, double eps) {
        double[] evals = new double[n];
        int i = 0;
        while (i < n) {
            if (i < n - 1 && Math.abs(T[i + 1][i]) > eps * (Math.abs(T[i][i]) + Math.abs(T[i + 1][i + 1]) + 1.0)) {
                double a = T[i][i], b = T[i][i + 1];
                double c = T[i + 1][i], d = T[i + 1][i + 1];
                double disc = (a - d) * (a - d) + 4.0 * b * c;
                if (disc < 0) {
                    evals[i] = (a + d) / 2.0;
                    evals[i + 1] = evals[i];
                } else {
                    double sd = Math.sqrt(disc);
                    evals[i]     = (a + d + sd) / 2.0;
                    evals[i + 1] = (a + d - sd) / 2.0;
                }
                i += 2;
            } else {
                evals[i] = T[i][i];
                i++;
            }
        }
        return evals;
    }

    private double[][] backSubstituteSchurEigenvectors(double[][] T, double[][] Q, int n, double eps) {
        double[] evals = extractRealSchurEigenvalues(T, n, eps);
        double[][] eigvecs = new double[n][n];

        int p = 0;
        while (p < n) {
            boolean is2x2 = (p < n - 1) &&
                Math.abs(T[p + 1][p]) > eps * (Math.abs(T[p][p]) + Math.abs(T[p + 1][p + 1]) + 1.0);

            if (is2x2) {
                double a = T[p][p], b = T[p][p + 1];
                double c = T[p + 1][p], d = T[p + 1][p + 1];
                double disc = (a - d) * (a - d) + 4.0 * b * c;

                double[] y1 = new double[n];
                double[] y2 = new double[n];

                if (disc < 0) {
                    y1[p] = 1.0; y1[p + 1] = 0.0;
                    y2[p] = 0.0; y2[p + 1] = 1.0;
                } else {
                    double sd = Math.sqrt(disc);
                    double lam1 = (a + d + sd) / 2.0;
                    double lam2 = (a + d - sd) / 2.0;
                    if (Math.abs(b) > Math.abs(c)) {
                        y1[p] = 1.0; y1[p + 1] = Math.abs(b) > 1e-30 ? (lam1 - a) / b : 0.0;
                        y2[p] = 1.0; y2[p + 1] = Math.abs(b) > 1e-30 ? (lam2 - a) / b : 0.0;
                    } else {
                        y1[p + 1] = 1.0; y1[p] = Math.abs(c) > 1e-30 ? (lam1 - d) / c : 0.0;
                        y2[p + 1] = 1.0; y2[p] = Math.abs(c) > 1e-30 ? (lam2 - d) / c : 0.0;
                    }
                }

                for (int k = p - 1; k >= 0; k--) {
                    double sum1 = 0.0, sum2 = 0.0;
                    for (int j = k + 1; j <= p + 1; j++) {
                        sum1 += T[k][j] * y1[j];
                        sum2 += T[k][j] * y2[j];
                    }
                    double denom1 = T[k][k] - evals[p];
                    double denom2 = T[k][k] - evals[p + 1];
                    y1[k] = (Math.abs(denom1) > eps) ? -sum1 / denom1 : 0.0;
                    y2[k] = (Math.abs(denom2) > eps) ? -sum2 / denom2 : 0.0;
                }

                for (int i = 0; i < n; i++) {
                    double s1 = 0.0, s2 = 0.0;
                    for (int j = 0; j < n; j++) {
                        s1 += Q[i][j] * y1[j];
                        s2 += Q[i][j] * y2[j];
                    }
                    eigvecs[i][p]     = s1;
                    eigvecs[i][p + 1] = s2;
                }
                p += 2;
            } else {
                double lam = T[p][p];
                double[] y = new double[n];
                y[p] = 1.0;
                for (int k = p - 1; k >= 0; k--) {
                    double sum = 0.0;
                    for (int j = k + 1; j <= p; j++) sum += T[k][j] * y[j];
                    double denom = T[k][k] - lam;
                    y[k] = (Math.abs(denom) > eps) ? -sum / denom : 0.0;
                }
                for (int i = 0; i < n; i++) {
                    double s = 0.0;
                    for (int j = 0; j < n; j++) s += Q[i][j] * y[j];
                    eigvecs[i][p] = s;
                }
                p++;
            }
        }
        return eigvecs;
    }
